Lưu ý điều hành

  • Đây là một buổi thực hành của một ứng dụng. Chỉ đánh giá bốn hàm sinh viên tự viết: merge_sort, binary_search, bfs_routes và knapsack. Danh sách, FIFO, ngăn xếp, BST, DFS, đệ quy và vét cạn đã cung cấp để sinh viên thấy sự phối hợp; không biến việc đọc mã hỗ trợ thành bảy đề mới.
  • Giữ thứ tự nhiệm vụ sort → search → BFS → DP vì build_session cần merge_sort để chuẩn hóa danh sách kề. Sau BFS, tái dùng merge_sort cho thứ tự ưu tiên theo lợi ích/τ. Nếu giới thiệu BFS trước bằng hình, dùng đồ thị mẫu đã cho và không chạy build_session khi hàm sort còn thiếu.
  • Trước giờ học, kiểm tra Julia chạy được, khung sinh viên mở được và chế độ --data tạo MINI001 khi bốn TODO còn thiếu. Chuẩn bị máy giảng viên có mã lời giải riêng để trình diễn, nhưng không đặt lời giải trong bộ sinh viên.
  • Phân biệt kết quả minh họa đã biết với kiểm thử chương trình. Một hoạt hình đẹp hoặc một menu mở được chưa chứng minh bốn hàm đúng. Chỉ ghi đạt sau chạy đúng pha kiểm thử và đối chiếu đầu ra thực tế.
  • Mốc hỗ trợ: sau NV1 cần dãy mới [1,2,3]; sau NV2 phải tra Order/nothing; sau NV3 cần distance và routes hợp lệ cùng costs=Dict(1=>3,2=>5,3=>7); sau NV4 mới chạy ứng dụng trọn luồng.
  • Bảng ánh xạ bảy chương là phụ lục giải thích vai trò; một buổi không hiện thực tất cả 29 bài của học phần. Mục tiêu là bốn thuật toán trọng tâm và sự nối kết đầu ra–đầu vào.
  • MSSV chỉ dùng phần phát triển sau bộ mini trong phương án 180 phút. Không ép sinh viên vừa học chỉ số bảng DP vừa xử lý ngay bộ ngẫu nhiên 8–10 đơn trước khi kiểm tra được ba đơn bằng tay.
  • Phần trình bày bằng Julia là bắt buộc. Trong lớp tập trung bốn hàm và kiểm thử; sau buổi học sinh viên hoàn thiện báo cáo cùng notebook Julia/Pluto mẫu gọi cùng lõi. Máy chưa cài Pluto được chạy trực tiếp tệp notebook .jl bằng Julia. Không tăng số TODO hoặc kéo dài kịch bản trên lớp.

merge_sort

function merge_sort(a::AbstractVector, before::Function)
    length(a) <= 1 && return collect(a)
    mid = div(length(a),2)
    left = merge_sort(a[1:mid],before)
    right = merge_sort(a[mid+1:end],before)
    out = eltype(a)[]
    i = j = 1
    while i <= length(left) && j <= length(right)
        if before(right[j],left[i])
            push!(out,right[j]); j += 1
        else
            push!(out,left[i]); i += 1
        end
    end
    while i <= length(left); push!(out,left[i]); i += 1; end
    while j <= length(right); push!(out,right[j]); j += 1; end
    out
end

binary_search

function binary_search(a::Vector{Order},id::Int)
    lo,hi = 1,length(a)
    while lo <= hi
        mid = lo + div(hi-lo,2)
        a[mid].id == id && return a[mid]
        if a[mid].id < id; lo = mid+1; else; hi = mid-1; end
    end
    nothing
end

function bfs_routes(graph,source=1)
    n = length(graph)
    distance = fill(-1,n); parent = zeros(Int,n)
    q = CircularQueue(n)
    distance[source] = 0; enqueue!(q,source)
    while q.count > 0
        u = dequeue!(q)
        for v in graph[u]
            if distance[v] == -1
                distance[v] = distance[u]+1
                parent[v] = u
                enqueue!(q,v) || error("Hàng đợi BFS quá tải.")
            end
        end
    end
    routes = [Int[] for _ in 1:n]
    for v in 1:n
        distance[v] == -1 && continue
        route = Int[]; cursor = v
        while cursor != 0
            push!(route,cursor); cursor = parent[cursor]
        end
        routes[v] = reverse(route)
    end
    (distance=distance,parent=parent,routes=routes)
end

knapsack

function knapsack(orders,costs,budget)
    budget >= 0 || throw(ArgumentError("Ngân sách không được âm."))
    a = merge_sort(orders,(x,y)->x.id < y.id)
    n = length(a); table = zeros(Int,n+1,budget+1)
    for i in 1:n, b in 0:budget
        o = a[i]; cost = costs[o.id]
        cost > 0 || throw(ArgumentError("Chi phí đơn phải dương."))
        skip = table[i,b+1]
        take = cost <= b ? table[i,b-cost+1] + o.benefit : -1
        table[i+1,b+1] = max(skip,take)
    end
    ids = Int[]; b = budget
    for i in n:-1:1
        if table[i+1,b+1] > table[i,b+1]
            push!(ids,a[i].id); b -= costs[a[i].id]
        end
    end
    ids = reverse(ids)
    (ids=ids,value=table[n+1,budget+1],cost=budget-b,table=table)
end
